    81→
    82→export function Header() {
    83→  const pathname = usePathname();
    84→  const [scrolled, setScrolled] = useState(false);
    85→  const [compact, setCompact] = useState(false);
    86→  const [dropdownOpen, setDropdownOpen] = useState(false);
    87→  const [mobileOpen, setMobileOpen] = useState(false);
    88→  const openTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    89→  const closeTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    90→  const megaMenuRef = useRef<HTMLDivElement>(null);
    91→  const triggerRef = useRef<HTMLDivElement>(null);
    92→
    93→  // Check if a product link matches current path
    94→  const isProductActive = pathname.startsWith('/products');
    95→
    96→  // Scroll detection
    97→  useEffect(() => {
    98→    const handleScroll = () => {
    99→      const y = window.scrollY;
   100→      setScrolled(y > 10);
   101→      setCompact(y > 80);
   102→    };
   103→    window.addEventListener('scroll', handleScroll, { passive: true });
   104→    return () => window.removeEventListener('scroll', handleScroll);
   105→  }, []);
   106→
   107→  // Close mobile menu on resize to desktop
   108→  // Use lg breakpoint (1024px) to match when the hamburger hides and
   109→  // desktop navigation appears.
   110→  useEffect(() => {
   111→    const handleResize = () => {
   112→      if (window.innerWidth >= 1024) {
   113→        setMobileOpen(false);
   114→      }
   115→    };
   116→    window.addEventListener('resize', handleResize);
   117→    return () => window.removeEventListener('resize', handleResize);
   118→  }, []);
   119→
   120→  // Body scroll lock when mobile menu is open.
   121→  // Radix Dialog (SheetContent) already handles overflow: hidden on <body>,
   122→  // but we also set padding-right equal to the scrollbar width to prevent
   123→  // the header from jumping horizontally when the scrollbar disappears.
   124→  useEffect(() => {
   125→    if (mobileOpen) {
   126→      const scrollbarWidth = window.innerWidth - document.documentElement.clientWidth;
   127→      document.body.style.overflow = 'hidden';
   128→      if (scrollbarWidth > 0) {
   129→        document.body.style.paddingRight = `${scrollbarWidth}px`;
   130→      }
   131→    } else {
   132→      document.body.style.overflow = '';
   133→      document.body.style.paddingRight = '';
   134→    }
   135→    return () => {
   136→      document.body.style.overflow = '';
   137→      document.body.style.paddingRight = '';
   138→    };
   139→  }, [mobileOpen]);
   140→
   141→  // Close dropdown on outside click
   142→  useEffect(() => {
   143→    if (!dropdownOpen) return;
   144→
   145→    const handleClickOutside = (e: MouseEvent) => {
   146→      if (
   147→        megaMenuRef.current &&
   148→        !megaMenuRef.current.contains(e.target as Node) &&
   149→        triggerRef.current &&
   150→        !triggerRef.current.contains(e.target as Node)
   151→      ) {
   152→        setDropdownOpen(false);
   153→      }
   154→    };
   155→
   156→    document.addEventListener('mousedown', handleClickOutside);
   157→    return () => document.removeEventListener('mousedown', handleClickOutside);
   158→  }, [dropdownOpen]);
   159→
   160→  // Hover-intent handlers with open/close delay
   161→  const clearOpenTimeout = useCallback(() => {
   162→    if (openTimeoutRef.current) {
   163→      clearTimeout(openTimeoutRef.current);
   164→      openTimeoutRef.current = null;
   165→    }
   166→  }, []);
   167→
   168→  const clearCloseTimeout = useCallback(() => {
   169→    if (closeTimeoutRef.current) {
   170→      clearTimeout(closeTimeoutRef.current);
   171→      closeTimeoutRef.current = null;
   172→    }
   173→  }, []);
   174→
   175→  const handleDropdownEnter = useCallback(() => {
   176→    clearCloseTimeout();
   177→    clearOpenTimeout();
   178→    openTimeoutRef.current = setTimeout(() => {
   179→      setDropdownOpen(true);
   180→    }, OPEN_DELAY);
   181→  }, [clearCloseTimeout, clearOpenTimeout]);
   182→
   183→  const handleDropdownLeave = useCallback(() => {
   184→    clearOpenTimeout();
   185→    clearCloseTimeout();
   186→    closeTimeoutRef.current = setTimeout(() => {
   187→      setDropdownOpen(false);
   188→    }, CLOSE_DELAY);
   189→  }, [clearOpenTimeout, clearCloseTimeout]);
   190→
   191→  // Keyboard handler for dropdown
   192→  const handleDropdownKeyDown = useCallback((e: React.KeyboardEvent) => {
   193→    if (e.key === ESCAPE_KEY) {
   194→      setDropdownOpen(false);
   195→      // Return focus to the trigger button
   196→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   197→      if (triggerBtn instanceof HTMLElement) {
   198→        triggerBtn.focus();
   199→      }
   200→    }
   201→    // Arrow Down opens the menu
   202→    if (e.key === 'ArrowDown' && !dropdownOpen) {
   203→      e.preventDefault();
   204→      setDropdownOpen(true);
   205→    }
   206→  }, [dropdownOpen]);
   207→
   208→  // Mega-menu keyboard navigation
   209→  const handleMegaMenuKeyDown = useCallback((e: React.KeyboardEvent) => {
   210→    if (e.key === ESCAPE_KEY) {
   211→      setDropdownOpen(false);
   212→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   213→      if (triggerBtn instanceof HTMLElement) {
   214→        triggerBtn.focus();
   215→      }
   216→    }
   217→    // Tab trap: focus stays within mega-menu while open
   218→    if (e.key === 'Tab' && megaMenuRef.current) {
   219→      const focusable = megaMenuRef.current.querySelectorAll<HTMLElement>(
   220→        'a[href], button, [tabindex]:not([tabindex="-1"])'
   221→      );
   222→      const first = focusable[0];
   223→      const last = focusable[focusable.length - 1];
   224→      if (e.shiftKey && document.activeElement === first) {
   225→        e.preventDefault();
   226→        last.focus();
   227→      } else if (!e.shiftKey && document.activeElement === last) {
   228→        e.preventDefault();
   229→        first.focus();
   230→      }
   231→    }
   232→  }, []);
   233→
   234→  // Close dropdown on route change
   235→  // Wrap in a transition callback to satisfy the lint rule about
   236→  // setState in an effect body — we sync the dropdown state to
   237→  // the pathname after navigation completes.
   238→  useEffect(() => {
   239→    // The pathname has changed, so we close the dropdown.
   240→    // This is synchronization with an external system (the router),
   241→    // which is a valid use of setState in an effect.
   242→    startTransition(() => {
   243→      setDropdownOpen(false);
   244→    });
   245→  }, [pathname]);
   246→
   247→  // Cleanup timeouts on unmount
   248→  useEffect(() => {
   249→    return () => {
   250→      clearOpenTimeout();
   251→      clearCloseTimeout();
   252→    };
   253→  }, [clearOpenTimeout, clearCloseTimeout]);
   254→
   255→  // Get the electrical insulation and waterproofing items
   256→  const electricalItems = productNavigationByCategory['electrical-insulation'];
   257→  const waterproofingItems = productNavigationByCategory['waterproofing-civil-protection'];
   258→
   259→  return (
   260→    <header className="relative z-50">
   261→      {/* ── Top Contact Strip (desktop only) ── */}
   262→      {/* Dark utility strip sits above the pure-white brand header to give
   263→          the logo zone maximum contrast and a premium two-tone feel. Text
   264→          uses be-charcoal-800 (#38383A) on be-white (#FFFFFF) — contrast
   265→          ≈ 10.3:1, passes WCAG AAA. Hover state uses be-yellow-text
   266→          (#755600, contrast ≈ 4.7:1) so links remain AA on hover.
   267→          Subtle bottom border (be-header-border #E8E2D8) separates from
   268→          the main white header bar. */}
   269→      <div className="hidden md:block bg-be-white border-b border-be-header-border">
   270→        <div className="container-site page-horizontal-padding flex items-center justify-between h-8">
   271→          {/* Left: contact info */}
   272→          <div className="flex items-center gap-5 text-metadata text-be-charcoal-800">
   273→            <a
   274→              href={`mailto:${company.email}`}
   275→              className="flex items-center gap-1.5 hover:text-be-yellow-text transition-colors"
   276→            >
   277→              <Mail className="size-3.5 text-be-charcoal-800" aria-hidden="true" focusable="false" />
   278→              <span>{company.email}</span>
   279→            </a>
   280→            <a
   281→              href={`tel:${company.phonePrimaryTel}`}
   282→              className="flex items-center gap-1.5 hover:text-be-yellow-text transition-colors"
   283→            >
   284→              <Phone className="size-3.5 text-be-charcoal-800" aria-hidden="true" focusable="false" />
   285→              <span>{company.phonePrimary}</span>
   286→            </a>
   287→          </div>
   288→
   289→          {/* Right: social / WhatsApp */}
   290→          <div className="flex items-center gap-3">
   291→            <a
   292→              href={company.whatsapp.href}
   293→              target="_blank"
   294→              rel="noopener noreferrer"
   295→              className="flex items-center gap-1.5 text-metadata text-be-charcoal-800 hover:text-be-yellow-text transition-colors"
   296→              aria-label="Chat on WhatsApp (opens in a new tab)"
   297→            >
   298→              <MessageCircle className="size-3.5 text-be-charcoal-800" aria-hidden="true" focusable="false" />
   299→              <span>{company.whatsapp.label}</span>
   300→            </a>
   301→          </div>
   302→        </div>
   303→      </div>
   304→
   305→      {/* ── Main Header Bar ── */}
   306→      {/* The sticky bar is the positioning context for the Products
   307→          mega-menu. Rendering the menu as a child of this sticky bar
   308→          (instead of inside the small Products trigger) lets us centre it
   309→          on the header container / viewport rather than on the trigger,
   310→          which previously caused left-clipping at narrower widths. */}
   311→      <div
   312→        className={cn(
   313→          'sticky top-0 z-50 bg-be-white border-b border-be-header-border transition-all duration-300',
   314→          scrolled && 'shadow-sm',
   315→          // Stable mobile height of 64px (h-16), with slightly taller
   316→          // desktop states to accommodate the contact strip above.
   317→          compact ? 'h-16 md:h-[72px]' : 'h-16 md:h-[84px]'
   318→        )}
   319→      >
   320→        {/* Responsive CSS Grid:
   321→            Mobile (<lg): 2-column layout — logo (flex) | hamburger (auto).
   322→            Desktop (lg+):  3-column layout — logo | nav | CTA.
   323→            minmax(190px,1fr) on both sides keeps the navigation
   324→            mathematically centred regardless of logo / CTA width. */}
   325→        <div className="container-site page-horizontal-padding grid grid-cols-[minmax(0,1fr)_auto] items-center h-full gap-3 lg:grid-cols-[minmax(190px,1fr)_auto_minmax(190px,1fr)] lg:gap-4">
   326→          {/* ── Column 1: Logo zone (left-aligned) ── */}
   327→          {/* Pure-white brand area — the logo sits directly on the white

   335→          <div className="flex items-center justify-start min-w-0">
   336→            <Link
   337→              href="/"
   338→              className="shrink-0 flex items-center px-2 sm:px-2.5 py-1.5 rounded-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-yellow-500 focus-visible:ring-offset-2 focus-visible:ring-offset-be-white transition-shadow"
   339→              aria-label="Bharat Electrosafe — Home"
   340→            >
   341→              {/* Intrinsic source dimensions are 1589x580 (aspect ≈ 2.74:1).
   342→                  Display sizes ramp from mobile (140px, top of the 130–145
   343→                  range) up to desktop (185px, within 170–190), with a
   344→                  compact state (138–160px, within 145–160 desktop) when
   345→                  scrolled. `priority` because the logo is above the fold

   347→                  Source is the lossless PNG master (188KB, 8-bit RGBA) —
   348→                  the sharpest available transparent asset — and Next.js
   349→                  Image re-optimizes per viewport width and pixel density

   351→                  gives maximum crispness with no client-weight penalty. */}
   352→              <Image
   353→                src="/images/brand/bharat-electrosafe-logo-full.png"
   354→                alt="Bharat Electrosafe logo"
   355→                width={1589}
   356→                height={580}
   357→                sizes="(max-width: 359px) 120px, (max-width: 429px) 138px, (max-width: 767px) 142px, (max-width: 1023px) 172px, 185px"
   358→                className={cn(
   359→                  'object-contain transition-all duration-300 h-auto w-[120px] sm:w-[128px] md:w-[142px] lg:w-[185px]',
   360→                  compact && 'w-[110px] sm:w-[125px] md:w-[132px] lg:w-[160px]'
   361→                )}
   362→                priority
   363→              />
   364→            </Link>
   365→            {/* Subtle vertical divider — visible from lg+ only.
   366→                Hidden on mobile and tablet to avoid crowding the logo
   367→                next to the hamburger. 32px tall, 1px warm-neutral
   368→                (#E6E0D4). Colour sits between pure white and be-grey-250
   369→                so it reads as a refined seam rather than a hard rule. */}
   370→            <div
   371→              className="hidden lg:block w-px h-8 bg-be-logo-divider ml-4 sm:ml-6 lg:ml-8"
   372→              aria-hidden="true"
   373→            />
   374→          </div>
   375→
   376→          {/* ── Column 2: Desktop Navigation (centred) ── */}
   377→          {/* Desktop nav shows at lg+ (≥1024px). At md (768-1023) the
   378→              horizontal budget is too tight for logo + 4 nav items + CTA,
   379→              so we keep the mobile sheet through tablet. */}
   380→          <nav className="hidden lg:flex items-center justify-center gap-1.5" aria-label="Main navigation">
   381→            <Link
   382→              href="/"
   383→              className={cn(
   384→                'px-3.5 py-2 text-sm font-medium transition-colors rounded-md hover:bg-be-yellow-50 hover:text-be-yellow-text-hover',
   385→                pathname === '/' ? 'text-be-yellow-text border-l-[3px] border-be-yellow-500 pl-3' : 'text-be-charcoal-800'
   386→              )}
   387→              aria-current={pathname === '/' ? 'page' : undefined}
   388→            >
   389→              Home
   390→            </Link>
   391→
   392→            {/* Products: text links to /products, chevron opens mega-menu.
   393→                The mega-menu itself is rendered below as a child of the

   397→              className="flex items-center"
   398→              onMouseEnter={handleDropdownEnter}
   399→              onMouseLeave={handleDropdownLeave}
   400→              onKeyDown={handleDropdownKeyDown}
   401→            >
   402→              {/* "Products" links to /products overview page */}
   403→              <Link

   405→                className={cn(
   406→                  'px-2 py-2 text-sm font-medium transition-colors rounded-md hover:bg-be-yellow-50 hover:text-be-yellow-text-hover',

   412→              >
   413→                Products

   415→
   416→              {/* Separate chevron button to open the mega-menu.

   419→              <button
   420→                type="button"
   421→                className={cn(
   422→                  'px-2 py-2 text-sm font-medium rounded-md transition-colors',
   423→                  dropdownOpen || isProductActive
   424→                    ? 'text-be-yellow-text bg-be-yellow-50'

   427→                aria-expanded={dropdownOpen}
   428→                aria-haspopup="true"
   429→                aria-controls="products-mega-menu"
   430→                aria-label={
   431→                  dropdownOpen
   432→                    ? 'Close products menu'
   433→                    : 'Open products menu'
   434→                }

   436→              >
   437→                <ChevronDown
   438→                  className={cn(
   439→                    'size-4 transition-transform duration-200',
   440→                    dropdownOpen && 'rotate-180'

   442→                  aria-hidden="true"

   444→                />
   445→              </button>
   446→            </div>

   452→                pathname === '/about-us' ? 'text-be-yellow-text border-l-[3px] border-be-yellow-500 pl-3' : 'text-be-charcoal-800'
   453→              )}
   454→              aria-current={pathname === '/about-us' ? 'page' : undefined}
   455→            >
   456→              About Us

   463→                pathname === '/contact-us' ? 'text-be-yellow-text border-l-[3px] border-be-yellow-500 pl-3' : 'text-be-charcoal-800'
   464→              )}
   465→              aria-current={pathname === '/contact-us' ? 'page' : undefined}

   471→          {/* ── Column 3: CTA + Mobile menu (right-aligned) ── */}

   476→            {/* Quote CTA — visually slightly less heavy than the logo so

   486→            {/* Mobile hamburger */}
   487→            <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>

   491→                  className="lg:hidden inline-flex items-center justify-center size-11 rounded-md text-be-charcoal-950 hover:bg-be-grey-150 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-yellow-500 focus-visible:ring-offset-2"
   492→                  aria-label={mobileOpen ? 'Close navigation menu' : 'Open navigation menu'}

   499→
   500→              <SheetContent side="right" className="w-[320px] sm:w-[360px] bg-be-white p-0" id="mobile-navigation-sheet">

   504→                      <Image
