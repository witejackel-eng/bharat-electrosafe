   105→/* ────────────────────────────────────────────
   106→   Header component
   107→   ──────────────────────────────────────────── */
   108→
   109→export function Header() {
   110→  const pathname = usePathname();
   111→  const [dropdownOpen, setDropdownOpen] = useState(false);
   112→  const [mobileOpen, setMobileOpen] = useState(false);
   113→  const [expandedMobileGroups, setExpandedMobileGroups] = useState<Set<string>>(new Set());
   114→  const [previewVisual, setPreviewVisual] = useState<ProductVisualRole>(DEFAULT_PREVIEW);
   115→  const [previewKey, setPreviewKey] = useState(0); // for crossfade animation
   116→  const openTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   117→  const closeTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   118→  const megaMenuRef = useRef<HTMLDivElement>(null);
   119→  const triggerRef = useRef<HTMLDivElement>(null);
   120→
   121→  const isProductActive = pathname.startsWith('/products');
   122→
   123→  // Close mobile menu on resize to desktop
   124→  useEffect(() => {
   125→    const handleResize = () => { if (window.innerWidth >= 1024) setMobileOpen(false); };
   126→    window.addEventListener('resize', handleResize);
   127→    return () => window.removeEventListener('resize', handleResize);
   128→  }, []);
   129→
   130→  // Body scroll lock when mobile menu is open
   131→  useEffect(() => {
   132→    if (mobileOpen) {
   133→      const scrollbarWidth = window.innerWidth - document.documentElement.clientWidth;
   134→      document.body.style.overflow = 'hidden';
   135→      if (scrollbarWidth > 0) document.body.style.paddingRight = `${scrollbarWidth}px`;
   136→    } else {
   137→      document.body.style.overflow = '';
   138→      document.body.style.paddingRight = '';
   139→    }
   140→    return () => { document.body.style.overflow = ''; document.body.style.paddingRight = ''; };
   141→  }, [mobileOpen]);
   142→
   143→  // Close dropdown on outside click
   144→  useEffect(() => {
   145→    if (!dropdownOpen) return;
   146→    const handleClickOutside = (e: MouseEvent) => {
   147→      if (
   148→        megaMenuRef.current && !megaMenuRef.current.contains(e.target as Node) &&
   149→        triggerRef.current && !triggerRef.current.contains(e.target as Node)
   150→      ) setDropdownOpen(false);
   151→    };
   152→    document.addEventListener('mousedown', handleClickOutside);
   153→    return () => document.removeEventListener('mousedown', handleClickOutside);
   154→  }, [dropdownOpen]);
   155→
   156→  // Hover-intent handlers
   157→  const clearOpenTimeout = useCallback(() => { if (openTimeoutRef.current) { clearTimeout(openTimeoutRef.current); openTimeoutRef.current = null; } }, []);
   158→  const clearCloseTimeout = useCallback(() => { if (closeTimeoutRef.current) { clearTimeout(closeTimeoutRef.current); closeTimeoutRef.current = null; } }, []);
   159→
   160→  const handleDropdownEnter = useCallback(() => {
   161→    clearCloseTimeout(); clearOpenTimeout();
   162→    openTimeoutRef.current = setTimeout(() => setDropdownOpen(true), OPEN_DELAY);
   163→  }, [clearCloseTimeout, clearOpenTimeout]);
   164→
   165→  const handleDropdownLeave = useCallback(() => {
   166→    clearOpenTimeout(); clearCloseTimeout();
   167→    closeTimeoutRef.current = setTimeout(() => setDropdownOpen(false), CLOSE_DELAY);
   168→  }, [clearOpenTimeout, clearCloseTimeout]);
   169→
   170→  // Keyboard handler for dropdown trigger
   171→  const handleDropdownKeyDown = useCallback((e: React.KeyboardEvent) => {
   172→    if (e.key === ESCAPE_KEY) {
   173→      setDropdownOpen(false);
   174→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   175→      if (triggerBtn instanceof HTMLElement) triggerBtn.focus();
   176→    }
   177→    if (e.key === 'ArrowDown' && !dropdownOpen) { e.preventDefault(); setDropdownOpen(true); }
   178→  }, [dropdownOpen]);
   179→
   180→  // Mega-menu keyboard navigation
   181→  const handleMegaMenuKeyDown = useCallback((e: React.KeyboardEvent) => {
   182→    if (e.key === ESCAPE_KEY) {
   183→      setDropdownOpen(false);
   184→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   185→      if (triggerBtn instanceof HTMLElement) triggerBtn.focus();
   186→    }
   187→    if (e.key === 'Tab' && megaMenuRef.current) {
   188→      const focusable = megaMenuRef.current.querySelectorAll<HTMLElement>('a[href], button, [tabindex]:not([tabindex="-1"])');
   189→      const first = focusable[0]; const last = focusable[focusable.length - 1];
   190→      if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
   191→      else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
   192→    }
   193→  }, []);
   194→
   195→  // Close dropdown on route change
   196→  useEffect(() => { startTransition(() => setDropdownOpen(false)); }, [pathname]);
   197→
   198→  // Cleanup timeouts on unmount
   199→  useEffect(() => { return () => { clearOpenTimeout(); clearCloseTimeout(); }; }, [clearOpenTimeout, clearCloseTimeout]);
   200→
   201→  // Toggle mobile group expansion
   202→  const toggleMobileGroup = useCallback((groupId: string) => {
   203→    setExpandedMobileGroups((prev) => { const next = new Set(prev); if (next.has(groupId)) next.delete(groupId); else next.add(groupId); return next; });
   204→  }, []);
   205→
   206→  // Split product groups for mega-menu layout
   207→  const primaryGroup = productNavGroups[0];
   208→  const secondaryGroups = productNavGroups.slice(1);
   209→  const domesticSub = (primaryGroup.children as ProductNavSubGroup[])[0];
   210→  const internationalSub = (primaryGroup.children as ProductNavSubGroup[])[1];
   211→
   212→  // Preview update on hover/focus
   213→  const updatePreview = useCallback((href: string) => {
   214→    const visual = previewMap[href];
   215→    if (visual && visual.src !== previewVisual.src) {
   216→      setPreviewVisual(visual);
   217→      setPreviewKey((k) => k + 1);
   218→    }
   219→  }, [previewVisual.src]);
   220→
   221→  // Group-level preview update (for hovering a category heading)
   222→  const updateGroupPreview = useCallback((groupId: string) => {
   223→    const visual = GROUP_PREVIEWS[groupId];
   224→    if (visual && visual.src !== previewVisual.src) {
   225→      setPreviewVisual(visual);
   226→      setPreviewKey((k) => k + 1);
   227→    }
   228→  }, [previewVisual.src]);
   229→
   230→  const resetPreview = useCallback(() => {
   231→    if (previewVisual.src !== DEFAULT_PREVIEW.src) {
   232→      setPreviewVisual(DEFAULT_PREVIEW);
   233→      setPreviewKey((k) => k + 1);
   234→    }
   235→  }, [previewVisual.src]);
   236→
   237→  return (
   238→    <header className="contents z-50">
   239→      {/* ── Main-Header Bar ── */}
   240→      <div
   241→        className="sticky top-0 z-50 be-header-navy h-16 lg:h-20"
   242→      >
   243→        {/* 3-column grid: Logo | Nav | CTA */}
   244→        <div className="container-site page-horizontal-padding grid grid-cols-[minmax(0,1fr)_auto] items-center h-full gap-2 lg:grid-cols-[minmax(180px,1fr)_auto_minmax(180px,1fr)] lg:gap-4">
   245→          {/* ── Column 1: Logo ── */}
   246→          <div className="flex items-center justify-start min-w-0">
   247→            <Link
   248→              href="/"
   249→              className="shrink-0 flex items-center py-1 lg:py-1 rounded-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-brand-yellow focus-visible:ring-offset-2 focus-visible:ring-offset-be-navy-800"
   250→              aria-label="Bharat Electrosafe — Home"
   251→            >
   252→              <Image
   253→                src={LOGO_SRC}
   254→                alt="Bharat Electrosafe® logo"
   255→                width={LOGO_WIDTH}
   256→                height={LOGO_HEIGHT}
   257→                sizes="(max-width: 1023px) 97px, 125px"
   258→                className="object-contain h-[53px] lg:h-[68px] w-auto"
   259→                priority
   260→              />
   261→            </Link>
   262→          </div>
   263→
   264→          {/* ── Column 2: Desktop Navigation ── */}
   265→          <nav className="hidden lg:flex items-center justify-center gap-1" aria-label="Main navigation">
   266→            <Link
   267→              href="/"
   268→              className={cn(
   269→                'px-3 py-2 text-sm font-medium transition-colors',
   270→                pathname === '/'
   271→                  ? 'text-be-brand-yellow'
   272→                  : 'text-white/80 hover:text-white'
   273→              )}
   274→              aria-current={pathname === '/' ? 'page' : undefined}
   275→            >
   276→              Home
   277→            </Link>
   278→
   279→            {/* Products: hover opens mega-menu */}
   280→            <div
   281→              ref={triggerRef}
   282→              className="flex items-center relative"
   283→              onMouseEnter={handleDropdownEnter}
   284→              onMouseLeave={handleDropdownLeave}
   285→              onKeyDown={handleDropdownKeyDown}
   286→            >
   287→              <Link
   288→                href="/products"
   289→                className={cn(
   290→                  'px-2 py-2 text-sm font-medium transition-colors',
   291→                  isProductActive && !dropdownOpen ? 'text-be-brand-yellow' : 'text-white/80 hover:text-white'
   292→                )}
   293→                aria-current={isProductActive ? 'page' : undefined}
   294→              >
   295→                Products
   296→              </Link>
   297→              <button
   298→                type="button"
   299→                className={cn(
   300→                  'px-1.5 py-2 text-sm font-medium transition-colors',
   301→                  dropdownOpen
   302→                    ? 'text-be-brand-yellow'
   303→                    : isProductActive
   304→                      ? 'text-be-brand-yellow'
   305→                      : 'text-white/80 hover:text-white'
   306→                )}
   307→                aria-expanded={dropdownOpen}
   308→                aria-haspopup="true"
   309→                aria-controls="products-mega-menu"
   310→                aria-label={dropdownOpen ? 'Close products menu' : 'Open products menu'}
   311→                onClick={() => setDropdownOpen((prev) => !prev)}
   312→              >
   313→                <ChevronDown
   314→                  className={cn('size-3.5 transition-transform duration-200', dropdownOpen && 'rotate-180')}
   315→                  aria-hidden="true"
   316→                  focusable="false"
   317→                />
   318→              </button>
   319→              {/* Subtle active indicator bar */}
   320→              {(dropdownOpen || isProductActive) && (
   321→                <span className="absolute -bottom-0.5 left-1/2 -translate-x-1/2 h-0.5 w-6 bg-be-brand-yellow rounded-full" aria-hidden="true" />
   322→              )}
   323→            </div>
   324→
   325→            <Link
   326→              href="/about-us"
   327→              className={cn(
   328→                'px-3 py-2 text-sm font-medium transition-colors',
   329→                pathname === '/about-us' ? 'text-be-brand-yellow' : 'text-white/80 hover:text-white'
   330→              )}
   331→              aria-current={pathname === '/about-us' ? 'page' : undefined}
   332→            >
   333→              About Us
   334→            </Link>
   335→
   336→            <Link
   337→              href="/contact-us"
   338→              className={cn(
   339→                'px-3 py-2 text-sm font-medium transition-colors',
   340→                pathname === '/contact-us' ? 'text-be-brand-yellow' : 'text-white/80 hover:text-white'
   341→              )}
   342→              aria-current={pathname === '/contact-us' ? 'page' : undefined}
   343→            >
   344→              Contact Us
   345→            </Link>
   346→          </nav>
   347→
   348→          {/* ── Column 3: CTA + Mobile menu ── */}
   349→          <div className="flex items-center justify-end gap-2 shrink-0 lg:shrink">
   350→            <PrimaryButton
   351→              href="/contact-us"
   352→              className="hidden lg:inline-flex text-sm px-4 py-2 shadow-sm hover:shadow-sm hover:translate-y-0 min-h-[40px]"
   353→            >
   354→              Request a Quote
   355→            </PrimaryButton>
   356→
   357→            {/* Mobile hamburger */}
   358→            <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>
   359→              <SheetTrigger asChild>
   360→                <button
   361→                  type="button"
   362→                  className="lg:hidden inline-flex items-center justify-center size-11 rounded-md text-white hover:bg-white/10 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-brand-yellow focus-visible:ring-offset-2 focus-visible:ring-offset-be-navy-800"
   363→                  aria-label={mobileOpen ? 'Close navigation menu' : 'Open navigation menu'}
   364→                  aria-expanded={mobileOpen}
   365→                  aria-controls="mobile-navigation-sheet"
   366→                >
   367→                  <Menu className="size-6" aria-hidden="true" focusable="false" />
   368→                </button>
   369→              </SheetTrigger>
   370→
   371→              <SheetContent side="right" className="w-[320px] sm:w-[360px] bg-be-navy-900 p-0 border-l border-white/10" id="mobile-navigation-sheet">
   372→                <SheetHeader className="px-5 pt-5 pb-3 border-b border-white/10">
   373→                  <SheetTitle className="flex items-center">
   374→                    <Link href="/" onClick={() => setMobileOpen(false)}>
   375→                      <Image
   376→                        src={LOGO_SRC}
   377→                        alt="Bharat Electrosafe® logo"
   378→                        width={LOGO_WIDTH}
   379→                        height={LOGO_HEIGHT}
   380→                        sizes="160px"
   381→                        className="object-contain w-[160px] h-auto"
   382→                      />
   383→                    </Link>
   384→                  </SheetTitle>
   385→                </SheetHeader>
   386→
   387→                <nav className="flex flex-col py-2 overflow-y-auto" aria-label="Mobile navigation">
   388→                  <Link href="/"
   389→                    className={cn('px-5 py-3.5 text-base font-medium transition-colors min-h-[44px] flex items-center', pathname === '/' ? 'text-be-brand-yellow bg-white/8 border-l-[3px] border-be-brand-yellow' : 'text-white/90 hover:bg-white/6 hover:text-be-brand-yellow')}
   390→                    onClick={() => setMobileOpen(false)} aria-current={pathname === '/' ? 'page' : undefined}
   391→                  >Home</Link>
   392→
   393→                  {/* Products accordion */}
   394→                  <Accordion type="single" collapsible className="px-0">
   395→                    <AccordionItem value="products" className="border-b-0">
   396→                      <AccordionTrigger className={cn('px-5 py-3.5 text-base font-medium transition-colors min-h-[44px] hover:no-underline', isProductActive ? 'text-be-brand-yellow bg-white/8 border-l-[3px] border-be-brand-yellow' : 'text-white/90 hover:bg-white/6 hover:text-be-brand-yellow')}>
   397→                        Products
   398→                      </AccordionTrigger>
   399→                      <AccordionContent className="pb-2">
   400→                        <div className="flex flex-col">
   401→                          {/* Electrical Insulating Mats — expanded by default */}
   402→                          <div className="pl-4">
   403→                            <div className="px-3 py-2 text-[0.75rem] font-semibold text-be-brand-yellow uppercase tracking-wider">
   404→                              Electrical Insulating Mats
   405→                            </div>
   406→                            {/* Domestic */}
   407→                            <div className="pl-3">
   408→                              <div className="px-2 py-1 text-[0.7rem] font-medium text-white/50 uppercase tracking-wider">
   409→                                Domestic Mats — IS 15652:2006
   410→                              </div>
   411→                              {domesticSub.items.map((item) => (
   412→                                <Link key={item.href} href={item.href}
   413→                                  className="flex items-center px-2 py-2.5 text-sm text-white/85 hover:text-be-brand-yellow hover:bg-white/6 transition-colors min-h-[44px]"
   414→                                  onClick={() => setMobileOpen(false)}
   415→                                >{item.name}</Link>
   416→                              ))}
   417→                            </div>
   418→                            {/* International */}
   419→                            <div className="pl-3 mt-1">
   420→                              <div className="px-2 py-1 text-[0.7rem] font-medium text-white/50 uppercase tracking-wider">
   421→                                International / Global — IEC 61111:2009
   422→                              </div>
   423→                              {internationalSub.items.map((item) => (
   424→                                <Link key={item.href} href={item.href}
   425→                                  className="flex items-center px-2 py-2.5 text-sm text-white/85 hover:text-be-brand-yellow hover:bg-white/6 transition-colors min-h-[44px]"
   426→                                  onClick={() => setMobileOpen(false)}
   427→                                >{item.name}</Link>
   428→                              ))}
   429→                            </div>
   430→                          </div>
   431→
   432→                          {/* Secondary groups */}
   433→                          {secondaryGroups.map((group) => {
   434→                            const items = group.hasSubGroups
   435→                              ? (group.children as ProductNavSubGroup[]).flatMap((s) => s.items)
   436→                              : (group.children as ProductNavLeaf[]);
   437→                            return (
   438→                              <div key={group.id} className="pl-4 mt-2">
   439→                                <div className="px-3 py-2 text-[0.75rem] font-semibold text-be-brand-yellow uppercase tracking-wider">
   440→                                  {group.name}
   441→                                </div>
   442→                                <div className="pl-3">
   443→                                  {items.map((item) => (
   444→                                    <Link key={item.href} href={item.href}
   445→                                      className="flex items-center px-2 py-2.5 text-sm text-white/85 hover:text-be-brand-yellow hover:bg-white/6 transition-colors min-h-[44px]"
   446→                                      onClick={() => setMobileOpen(false)}
   447→                                    >{item.name}</Link>
   448→                                  ))}
   449→                                </div>
   450→                              </div>
   451→                            );
   452→                          })}
   453→
   454→                          {/* View All + Technical Guidance */}
   455→                          <div className="mt-2 mx-5 border-t border-white/10 pt-3 space-y-1">
   456→                            <Link href="/products"
   457→                              className="flex items-center justify-between px-3 py-2.5 text-sm font-semibold text-be-brand-yellow hover:bg-white/6 rounded-md transition-colors min-h-[44px]"
   458→                              onClick={() => setMobileOpen(false)}
   459→                            >
   460→                              View All Products
   461→                              <ChevronRight className="size-3.5" aria-hidden="true" />
   462→                            </Link>
   463→                            <Link href="/contact-us?type=technical-guidance"
   464→                              className="flex items-center justify-between px-3 py-2.5 text-sm font-medium text-white/80 hover:text-be-brand-yellow hover:bg-white/6 rounded-md transition-colors min-h-[44px]"
   465→                              onClick={() => setMobileOpen(false)}
   466→                            >
   467→                              Technical Guidance
   468→                              <ChevronRight className="size-3.5" aria-hidden="true" />
   469→                            </Link>
   470→                          </div>
   471→                        </div>
   472→                      </AccordionContent>
   473→                    </AccordionItem>
   474→                  </Accordion>
   475→
   476→                  <div className="mx-5 my-1 h-px bg-white/10" />
   477→
   478→                  <Link href="/about-us"
   479→                    className={cn('px-5 py-3.5 text-base font-medium transition-colors min-h-[44px] flex items-center', pathname === '/about-us' ? 'text-be-brand-yellow bg-white/8 border-l-[3px] border-be-brand-yellow' : 'text-white/90 hover:bg-white/6 hover:text-be-brand-yellow')}
   480→                    onClick={() => setMobileOpen(false)} aria-current={pathname === '/about-us' ? 'page' : undefined}
   481→                  >About Us</Link>
