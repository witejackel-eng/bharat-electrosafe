    82→    82→export function Header() {
    83→    83→  const pathname = usePathname();

    85→    85→  const [compact, setCompact] = useState(false);
    86→    86→  const [dropdownOpen, setDropdownOpen] = useState(false);
    87→    87→  const [mobileOpen, setMobileOpen] = useState(false);
    88→    88→  const openTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    89→    89→  const closeTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    90→    90→  const megaMenuRef = useRef<HTMLDivElement>(null);
    91→    91→  const triggerRef = useRef<HTMLDivElement>(null);
    92→    92→
    93→    93→  // Check if a product link matches current path
    94→    94→  const isProductActive = pathname.startsWith('/products');
    95→    95→
    96→    96→  // Scroll detection
    97→    97→  useEffect(() => {
    98→    98→    const handleScroll = () => {
    99→    99→      const y = window.scrollY;
   100→   100→      setScrolled(y > 10);
   101→   101→      setCompact(y > 80);
   102→   102→    };
   103→   103→    window.addEventListener('scroll', handleScroll, { passive: true });
   104→   104→    return () => window.removeEventListener('scroll', handleScroll);
   105→   105→  }, []);
   106→   106→
   107→   107→  // Close mobile menu on resize to desktop
   108→   108→  // Use lg breakpoint (1024px) to match when the hamburger hides and
   109→   109→  // desktop navigation appears.
   110→   110→  useEffect(() => {
   111→   111→    const handleResize = () => {
   112→   112→      if (window.innerWidth >= 1024) {
   113→   113→        setMobileOpen(false);
   114→   114→      }
   115→   115→    };
   116→   116→    window.addEventListener('resize', handleResize);
   117→   117→    return () => window.removeEventListener('resize', handleResize);
   118→   118→  }, []);
   119→   119→
   120→   120→  // Body scroll lock when mobile menu is open.
   121→   121→  // Radix Dialog (SheetContent) already handles overflow: hidden on <body>,
   122→   122→  // but we also set padding-right equal to the scrollbar width to prevent
   123→   123→  // the header from jumping horizontally when the scrollbar disappears.
   124→   124→  useEffect(() => {
   125→   125→    if (mobileOpen) {
   126→   126→      const scrollbarWidth = window.innerWidth - document.documentElement.clientWidth;
   127→   127→      document.body.style.overflow = 'hidden';
   128→   128→      if (scrollbarWidth > 0) {
   129→   129→        document.body.style.paddingRight = `${scrollbarWidth}px`;
   130→   130→      }
   131→   131→    } else {
   132→   132→      document.body.style.overflow = '';
   133→   133→      document.body.style.paddingRight = '';
   134→   134→    }
   135→   135→    return () => {
   136→   136→      document.body.style.overflow = '';
   137→   137→      document.body.style.paddingRight = '';
   138→   138→    };
   139→   139→  }, [mobileOpen]);
   140→   140→
   141→   141→  // Close dropdown on outside click

   143→   143→    if (!dropdownOpen) return;

   146→   146→      if (
   147→   147→        megaMenuRef.current &&
   148→   148→        !megaMenuRef.current.contains(e.target as Node) &&
   149→   149→        triggerRef.current &&
   150→   150→        !triggerRef.current.contains(e.target as Node)
   151→   151→      ) {
   152→   152→        setDropdownOpen(false);
   153→   153→      }
   154→   154→    };
   155→   155→
   156→   156→    document.addEventListener('mousedown', handleClickOutside);
   157→   157→    return () => document.removeEventListener('mousedown', handleClickOutside);
   158→   158→  }, [dropdownOpen]);
   159→   159→
   160→   160→  // Hover-intent handlers with open/close delay
   161→   161→  const clearOpenTimeout = useCallback(() => {
   162→   162→    if (openTimeoutRef.current) {
   163→   163→      clearTimeout(openTimeoutRef.current);
   164→   164→      openTimeoutRef.current = null;
   165→   165→    }
   166→   166→  }, []);
   167→   167→
   168→   168→  const clearCloseTimeout = useCallback(() => {
   169→   169→    if (closeTimeoutRef.current) {
   170→   170→      clearTimeout(closeTimeoutRef.current);
   171→   171→      closeTimeoutRef.current = null;
   172→   172→    }
   173→   173→  }, []);
   174→   174→
   175→   175→  const handleDropdownEnter = useCallback(() => {
   176→   176→    clearCloseTimeout();
   177→   177→    clearOpenTimeout();
   178→   178→    openTimeoutRef.current = setTimeout(() => {
   179→   179→      setDropdownOpen(true);
   180→   180→    }, OPEN_DELAY);
   181→   181→  }, [clearCloseTimeout, clearOpenTimeout]);
   182→   182→
   183→   183→  const handleDropdownLeave = useCallback(() => {
   184→   184→    clearOpenTimeout();
   185→   185→    clearCloseTimeout();
   186→   186→    closeTimeoutRef.current = setTimeout(() => {
   187→   187→      setDropdownOpen(false);
   188→   188→    }, CLOSE_DELAY);
   189→   189→  }, [clearOpenTimeout, clearCloseTimeout]);
   190→   190→
   191→   191→  // Keyboard handler for dropdown
   192→   192→  const handleDropdownKeyDown = useCallback((e: React.KeyboardEvent) => {
   193→   193→    if (e.key === ESCAPE_KEY) {
   194→   194→      setDropdownOpen(false);
   195→   195→      // Return focus to the trigger button
   196→   196→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');

   198→   198→        triggerBtn.focus();
   199→   199→      }
   200→   200→    }
   201→   201→    // Arrow Down opens the menu
   202→   202→    if (e.key === 'ArrowDown' && !dropdownOpen) {
   203→   203→      e.preventDefault();
   204→   204→      setDropdownOpen(true);
   205→   205→    }
   206→   206→  }, [dropdownOpen]);
   207→   207→
   208→   208→  // Mega-menu keyboard navigation
   209→   209→  const handleMegaMenuKeyDown = useCallback((e: React.KeyboardEvent) => {
   210→   210→    if (e.key === ESCAPE_KEY) {
   211→   211→      setDropdownOpen(false);
   212→   212→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   213→   213→      if (triggerBtn instanceof HTMLElement) {
   214→   214→        triggerBtn.focus();
   215→   215→      }
   216→   216→    }
   217→   217→    // Tab trap: focus stays within mega-menu while open
   218→   218→    if (e.key === 'Tab' && megaMenuRef.current) {
   219→   219→      const focusable = megaMenuRef.current.querySelectorAll<HTMLElement>(
   220→   220→        'a[href], button, [tabindex]:not([tabindex="-1"])'
   221→   221→      );
   222→   222→      const first = focusable[0];
   223→   223→      const last = focusable[focusable.length - 1];
   224→   224→      if (e.shiftKey && document.activeElement === first) {
   225→   225→        e.preventDefault();
   226→   226→        last.focus();
   227→   227→      } else if (!e.shiftKey && document.activeElement === last) {
   228→   228→        e.preventDefault();
   229→   229→        first.focus();
   230→   230→      }
   231→   231→    }
   232→   232→  }, []);
   233→   233→
   234→   234→  // Close dropdown on route change

   237→   237→  // the pathname after navigation completes.

   241→   241→    // which is a valid use of setState in an effect.
   242→   242→    startTransition(() => {
   243→   243→      setDropdownOpen(false);
   244→   244→    });
   245→   245→  }, [pathname]);
   246→   246→
   247→   247→  // Cleanup timeouts on unmount
   248→   248→  useEffect(() => {
   249→   249→    return () => {
   250→   250→      clearOpenTimeout();
   251→   251→      clearCloseTimeout();
   252→   252→    };
   253→   253→  }, [clearOpenTimeout, clearCloseTimeout]);
   254→   254→
   255→   255→  // Get the electrical insulation and waterproofing items

   268→   268→          the main white header bar. */}
   269→   269→      <div className="hidden md:block bg-be-white border-b border-be-header-border">
   270→   270→        <div className="container-site page-horizontal-padding flex items-center justify-between h-8">
   271→   271→          {/* Left: contact info */}
   272→   272→          <div className="flex items-center gap-5 text-metadata text-be-charcoal-800">
   273→   273→            <a
   274→   274→              href={`mailto:${company.email}`}
