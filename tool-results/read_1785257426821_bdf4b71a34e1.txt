     1→/**
     2→ * Product Data — Bharat Electrosafe
     3→ *
     4→ * Comprehensive data definitions for all 6 product families.
     5→ *
     6→ * Every technical figure below is transcribed from the client's own product
     7→ * pages on bharatelectrosafe.com. The four insulating-mat families share one
     8→ * published specification table (IS 15652:2006, codes BES1001–BES1003), so
     9→ * that table is defined once and reused rather than re-typed per product —

    13→ * publishes no figure, no figure is invented.

    68→/**
    69→ * What a product image is *for*. Recorded per image so no component has to
    70→ * infer a role from a filename or from which array an image happens to sit in.
    71→ */
    72→export type ProductImageType =
    73→  | 'product'
    74→  | 'alternate'
    75→  | 'surface'
    76→  | 'marking'
    77→  | 'profile'
    78→  | 'installation'
    79→  | 'application';
    80→
    81→/**
    82→ * One placed image.
    83→ *
    84→ * `fit` is data, not a guess. An isolated product shot is `contain`, so a
    85→ * moulded marking, a coloured strip or a cut edge is never cropped off by the
    86→ * frame's aspect ratio; a photograph taken in a real setting is `cover`, so it
    87→ * fills the frame instead of floating on a letterbox. `position` moves the
    88→ * crop's focal point when the subject is not centred.
    89→ */
    90→export interface ProductImage {
    91→  src: string;
    92→  /** Describes what is visible. Never a keyword list, never "product image". */
    93→  alt: string;
    94→  fit: 'contain' | 'cover';
    95→  /** CSS object-position, e.g. 'center 30%'. Only meaningful with `cover`. */
    96→  position?: string;
    97→}
    98→
    99→export interface ProductGalleryImage extends ProductImage {
   100→  type: ProductImageType;
   101→  /** Short line explaining what the buyer is looking at. Not a repeat of alt. */
   102→  caption?: string;
   103→}
   104→
   105→/**
   106→ * Real, locally-stored imagery for a product.
   107→ *
   108→ * Every path resolves to a file under /public/media/products, written by
   109→ * scripts/build-product-gallery.mjs from a genuine client-owned source.
   110→ * Components read images from here rather than assembling their own arrays,
   111→ * so a product shows the same approved photograph everywhere it appears.
   112→ *
   113→ * `gallery` is the ordered story: best complete view first, then alternates,
   114→ * surface detail, markings and finally the product in context. `gallery[0]`
   115→ * *is* the product-page hero — it is not stored twice.
   116→ *
   117→ * `overview` and `application` point at entries already in `gallery`, so a
   118→ * caption or a correction is written once.
   119→ */
   120→export interface ProductImages {
   121→  /** Homepage card, /products tile, related-product tile, mega-menu. */
   122→  thumbnail: ProductImage;
   123→  gallery: ProductGalleryImage[];
   124→  /** Wider or contextual view shown beside the overview copy. */
   125→  overview?: ProductGalleryImage;
   126→  /** Genuine in-situ photograph, where one exists. */
   127→  application?: ProductGalleryImage;
   128→}
