   133→      setDropdownOpen(false);
   134→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   135→      if (triggerBtn instanceof HTMLElement) triggerBtn.focus();
   136→    }

   144→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');

   152→    }
