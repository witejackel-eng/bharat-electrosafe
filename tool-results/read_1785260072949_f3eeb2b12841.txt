    76→/* ────────────────────────────────────────────

    79→
    80→export function Header() {
    81→  const pathname = usePathname();
    82→  const [compact, setCompact] = useState(false);
    83→  const [dropdownOpen, setDropdownOpen] = useState(false);
    84→  const [mobileOpen, setMobileOpen] = useState(false);
    85→  const openTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    86→  const closeTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    87→  const megaMenuRef = useRef<HTMLDivElement>(null);
    88→  const triggerRef = useRef<HTMLDivElement>(null);
    89→
    90→  // Check if a product link matches current path
    91→  const isProductActive = pathname.startsWith('/products');
    92→
    93→  // Scroll detection
    94→  useEffect(() => {
    95→    const handleScroll = () => {
    96→      const y = window.scrollY;
    97→      setCompact(y > 80);
    98→    };
    99→    window.addEventListener('scroll', handleScroll, { passive: true });
   100→    return () => window.removeEventListener('scroll', handleScroll);
   101→  }, []);
   102→
   103→  // Close mobile menu on resize to desktop
   104→  // Use lg breakpoint (1024px) to match when the hamburger hides and
   105→  // desktop navigation appears.
   106→  useEffect(() => {
   107→    const handleResize = () => {
   108→      if (window.innerWidth >= 1024) {
   109→        setMobileOpen(false);
   110→      }
   111→    };
   112→    window.addEventListener('resize', handleResize);
   113→    return () => window.removeEventListener('resize', handleResize);
   114→  }, []);
   115→
   116→  // Body scroll lock when mobile menu is open.
   117→  // Radix Dialog (SheetContent) already handles overflow: hidden on <body>,
   118→  // but we also set padding-right equal to the scrollbar width to prevent
   119→  // the header from jumping horizontally when the scrollbar disappears.
   120→  useEffect(() => {
   121→    if (mobileOpen) {
   122→      const scrollbarWidth = window.innerWidth - document.documentElement.clientWidth;
   123→      document.body.style.overflow = 'hidden';
   124→      if (scrollbarWidth > 0) {
   125→        document.body.style.paddingRight = `${scrollbarWidth}px`;
   126→      }
   127→    } else {
   128→      document.body.style.overflow = '';
   129→      document.body.style.paddingRight = '';
   130→    }
   131→    return () => {
   132→      document.body.style.overflow = '';
   133→      document.body.style.paddingRight = '';
   134→    };
   135→  }, [mobileOpen]);
   136→
   137→  // Close dropdown on outside click
   138→  useEffect(() => {
   139→    if (!dropdownOpen) return;
   140→
   141→    const handleClickOutside = (e: MouseEvent) => {
   142→      if (
   143→        megaMenuRef.current &&
   144→        !megaMenuRef.current.contains(e.target as Node) &&
   145→        triggerRef.current &&
   146→        !triggerRef.current.contains(e.target as Node)
   147→      ) {
   148→        setDropdownOpen(false);
   149→      }
   150→    };
   151→
   152→    document.addEventListener('mousedown', handleClickOutside);
   153→    return () => document.removeEventListener('mousedown', handleClickOutside);
   154→  }, [dropdownOpen]);
   155→
   156→  // Hover-intent handlers with open/close delay
   157→  const clearOpenTimeout = useCallback(() => {
   158→    if (openTimeoutRef.current) {
   159→      clearTimeout(openTimeoutRef.current);
   160→      openTimeoutRef.current = null;
   161→    }
   162→  }, []);
   163→
   164→  const clearCloseTimeout = useCallback(() => {
   165→    if (closeTimeoutRef.current) {
   166→      clearTimeout(closeTimeoutRef.current);
   167→      closeTimeoutRef.current = null;
   168→    }
   169→  }, []);
   170→
   171→  const handleDropdownEnter = useCallback(() => {
   172→    clearCloseTimeout();
   173→    clearOpenTimeout();
   174→    openTimeoutRef.current = setTimeout(() => {
   175→      setDropdownOpen(true);
   176→    }, OPEN_DELAY);
   177→  }, [clearCloseTimeout, clearOpenTimeout]);
   178→
   179→  const handleDropdownLeave = useCallback(() => {
   180→    clearOpenTimeout();
   181→    clearCloseTimeout();
   182→    closeTimeoutRef.current = setTimeout(() => {
   183→      setDropdownOpen(false);
   184→    }, CLOSE_DELAY);
   185→  }, [clearOpenTimeout, clearCloseTimeout]);
   186→
   187→  // Keyboard handler for dropdown
   188→  const handleDropdownKeyDown = useCallback((e: React.KeyboardEvent) => {
   189→    if (e.key === ESCAPE_KEY) {
   190→      setDropdownOpen(false);
   191→      // Return focus to the trigger button
   192→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   193→      if (triggerBtn instanceof HTMLElement) {
   194→        triggerBtn.focus();
   195→      }
   196→    }
   197→    // Arrow Down opens the menu
   198→    if (e.key === 'ArrowDown' && !dropdownOpen) {
   199→      e.preventDefault();
   200→      setDropdownOpen(true);
   201→    }
   202→  }, [dropdownOpen]);
   203→
   204→  // Mega-menu keyboard navigation
   205→  const handleMegaMenuKeyDown = useCallback((e: React.KeyboardEvent) => {
   206→    if (e.key === ESCAPE_KEY) {
   207→      setDropdownOpen(false);
   208→      const triggerBtn = triggerRef.current?.querySelector('[aria-haspopup]');
   209→      if (triggerBtn instanceof HTMLElement) {
   210→        triggerBtn.focus();
   211→      }
   212→    }
   213→    // Tab trap: focus stays within mega-menu while open
   214→    if (e.key === 'Tab' && megaMenuRef.current) {
   215→      const focusable = megaMenuRef.current.querySelectorAll<HTMLElement>(
   216→        'a[href], button, [tabindex]:not([tabindex="-1"])'
   217→      );
   218→      const first = focusable[0];
   219→      const last = focusable[focusable.length - 1];
   220→      if (e.shiftKey && document.activeElement === first) {
   221→        e.preventDefault();
   222→        last.focus();
   223→      } else if (!e.shiftKey && document.activeElement === last) {
   224→        e.preventDefault();
   225→        first.focus();
   226→      }
   227→    }
   228→  }, []);
   229→
   230→  // Close dropdown on route change
   231→  // Wrap in a transition callback to satisfy the lint rule about
   232→  // setState in an effect body — we sync the dropdown state to
   233→  // the pathname after navigation completes.
   234→  useEffect(() => {
   235→    // The pathname has changed, so we close the dropdown.
   236→    // This is synchronization with an external system (the router),
   237→    // which is a valid use of setState in an effect.
   238→    startTransition(() => {
   239→      setDropdownOpen(false);
   240→    });
   241→  }, [pathname]);
   242→
   243→  // Cleanup timeouts on unmount
   244→  useEffect(() => {
   245→    return () => {
   246→      clearOpenTimeout();
   247→      clearCloseTimeout();
   248→    };
   249→  }, [clearOpenTimeout, clearCloseTimeout]);
   250→
   251→  // Get the electrical insulation and waterproofing items
   252→  const electricalItems = productNavigationByCategory['electrical-insulation'];
   253→  const waterproofingItems = productNavigationByCategory['waterproofing-civil-protection'];

   255→  return (
   256→    <header className="relative z-50">
   257→      {/* ── Top Contact Strip (desktop only) ── */}
   258→      {/* Navy utility strip continues the header's navy band upward, so the
   259→          logo zone and contact strip read as one designed composition rather
   260→          than a white strip floating above a navy bar. Text uses
   261→          rgba(255,255,255,0.78) on navy (#00275B) — contrast ≈ 7.5:1, passes
   262→          WCAG AAA. Hover state uses brand-yellow (#F4C313, contrast ≈ 8.6:1
   263→          on navy) for a bright, on-brand interaction. */}
   264→      <div className="hidden md:block be-contact-strip-navy">
   265→        <div className="container-site page-horizontal-padding flex items-center justify-between h-8">
   266→          {/* Left: contact info */}
   267→          <div className="flex items-center gap-5 text-metadata text-white/78">
   268→            <a
   269→              href={`mailto:${company.email}`}
   270→              className="flex items-center gap-1.5 hover:text-be-brand-yellow transition-colors"
   271→            >
   272→              <Mail className="size-3.5 text-white/70" aria-hidden="true" focusable="false" />
   273→              <span>{company.email}</span>
   274→            </a>
   275→            <a
   276→              href={`tel:${company.phonePrimaryTel}`}
   277→              className="flex items-center gap-1.5 hover:text-be-brand-yellow transition-colors"
   278→            >
   279→              <Phone className="size-3.5 text-white/70" aria-hidden="true" focusable="false" />
   280→              <span>{company.phonePrimary}</span>
   281→            </a>
   282→          </div>
   283→
   284→          {/* Right: social / WhatsApp */}
   285→          <div className="flex items-center gap-3">
   286→            <a
   287→              href={company.whatsapp.href}
   288→              target="_blank"
   289→              rel="noopener noreferrer"
   290→              className="flex items-center gap-1.5 text-metadata text-white/78 hover:text-be-brand-yellow transition-colors"
   291→              aria-label="Chat on WhatsApp (opens in a new tab)"
   292→            >
   293→              <MessageCircle className="size-3.5 text-white/70" aria-hidden="true" focusable="false" />
   294→              <span>{company.whatsapp.label}</span>
   295→            </a>
   296→          </div>
   297→        </div>
   298→      </div>
   299→
   300→      {/* ── Main Header Bar ── */}
   301→      {/* The sticky bar is the positioning context for the Products
   302→          mega-menu. Rendering the menu as a child of this sticky bar
   303→          (instead of inside the small Products trigger) lets us centre it
   304→          on the header container / viewport rather than on the trigger,
   305→          which previously caused left-clipping at narrower widths. */}
   306→      <div
   307→        className={cn(
   308→          'sticky top-0 z-50 be-header-navy transition-all duration-300',
   309→          compact && 'be-header-navy-compact',
   310→          // Stable mobile height of 64px (h-16), with slightly taller
   311→          // desktop states to accommodate the contact strip above.
   312→          compact ? 'h-16 md:h-[72px]' : 'h-16 md:h-[84px]'
   313→        )}
   314→      >
   315→        {/* Responsive CSS Grid:
   316→            Mobile (<lg): 2-column layout — logo (flex) | hamburger (auto).
   317→            Desktop (lg+):  3-column layout — logo | nav | CTA.
   318→            minmax(190px,1fr) on both sides keeps the navigation
   319→            mathematically centred regardless of logo / CTA width. */}
   320→        <div className="container-site page-horizontal-padding grid grid-cols-[minmax(0,1fr)_auto] items-center h-full gap-3 lg:grid-cols-[minmax(190px,1fr)_auto_minmax(190px,1fr)] lg:gap-4">
   321→          {/* ── Column 1: Logo zone (left-aligned) ── */}
   322→          {/* The transparent header logo (extracted from the client's master

   327→          <div className="flex items-center justify-start min-w-0">

   331→              aria-label="Bharat Electrosafe — Home"
   332→            >
   333→              {/* Transparent header logo — 891x349 (aspect ≈ 2.55:1).
   334→                  Display sizes ramp from mobile (132px) up to desktop
   335→                  (196px), with a compact state when scrolled. `priority`

   339→                src="/brand/bharat-electrosafe-logo-header.webp"
   340→                alt="Bharat Electrosafe logo"
   341→                width={891}
   342→                height={349}
   343→                sizes="(max-width: 359px) 124px, (max-width: 429px) 132px, (max-width: 767px) 140px, (max-width: 1023px) 168px, 196px"
   344→                className={cn(
   345→                  'object-contain transition-all duration-300 h-auto w-[124px] sm:w-[132px] md:w-[140px] lg:w-[196px]',
   346→                  compact && 'w-[116px] sm:w-[124px] md:w-[132px] lg:w-[168px]'
   347→                )}
   348→                priority
   349→              />
   350→            </Link>
   351→            {/* Subtle vertical divider — visible from lg+ only.
   352→                A faint white divider replaces the old warm-neutral seam,
   353→                matching the navy composition. */}
   354→            <div
   355→              className="hidden lg:block w-px h-8 bg-white/12 ml-4 sm:ml-6 lg:ml-8"
   356→              aria-hidden="true"
   357→            />
   358→          </div>
   359→
   360→          {/* ── Column 2: Desktop Navigation (centred) ── */}
   361→          {/* Desktop nav shows at lg+ (≥1024px). At md (768-1023) the
   362→              horizontal budget is too tight for logo + 4 nav items + CTA,

   364→          <nav className="hidden lg:flex items-center justify-center gap-1.5" aria-label="Main navigation">
   365→            <Link
   366→              href="/"
   367→              className={cn(
   368→                'relative px-3.5 py-2 text-sm font-medium rounded-md be-nav-link',
   369→                pathname === '/' && 'be-nav-link-active'
   370→              )}
   371→              aria-current={pathname === '/' ? 'page' : undefined}
   372→            >
   373→              Home

   375→
   376→            {/* Products: text links to /products, chevron opens mega-menu.

   381→              className="flex items-center"
   382→              onMouseEnter={handleDropdownEnter}
   383→              onMouseLeave={handleDropdownLeave}
   384→              onKeyDown={handleDropdownKeyDown}
   385→            >
   386→              {/* "Products" links to /products overview page */}
   387→              <Link
   388→                href="/products"
   389→                className={cn(
   390→                  'relative px-2 py-2 text-sm font-medium rounded-md be-nav-link',
   391→                  isProductActive && 'be-nav-link-active'
   392→                )}
   393→                aria-current={isProductActive ? 'page' : undefined}
   394→              >
   395→                Products
   396→              </Link>
   397→
   398→              {/* Separate chevron button to open the mega-menu.

   401→              <button
   402→                type="button"
   403→                className={cn(
   404→                  'px-2 py-2 text-sm font-medium rounded-md transition-colors',
   405→                  dropdownOpen || isProductActive
   406→                    ? 'text-be-brand-yellow bg-white/8'
   407→                    : 'be-nav-link'
   408→                )}
   409→                aria-expanded={dropdownOpen}
   410→                aria-haspopup="true"
   411→                aria-controls="products-mega-menu"
   412→                aria-label={
   413→                  dropdownOpen
   414→                    ? 'Close products menu'
   415→                    : 'Open products menu'

   419→                <ChevronDown

   421→                    'size-4 transition-transform duration-200',
   422→                    dropdownOpen && 'rotate-180'
   423→                  )}
   424→                  aria-hidden="true"
   425→                  focusable="false"
   426→                />
   427→              </button>
   428→            </div>
   429→
   430→            <Link
   431→              href="/about-us"
   432→              className={cn(
   433→                'relative px-3.5 py-2 text-sm font-medium rounded-md be-nav-link',

   455→              by the logo. justify-self-end right-aligns the hamburger

   458→            {/* Quote CTA — yellow fill with navy text for maximum contrast

   462→              className="hidden lg:inline-flex text-sm px-4 py-2 shadow-sm hover:shadow-sm hover:translate-y-0"
   463→            >

   471→                  type="button"
   472→                  className="lg:hidden inline-flex items-center justify-center size-11 rounded-md text-white hover:bg-white/10 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-brand-yellow focus-visible:ring-offset-2 focus-visible:ring-offset-be-navy-800"
   473→                  aria-label={mobileOpen ? 'Close navigation menu' : 'Open navigation menu'}
   474→                  aria-expanded={mobileOpen}

   492→                      />
   493→                    </Link>
   494→                  </SheetTitle>
   495→                </SheetHeader>
   496→
   497→                <nav className="flex flex-col py-2 overflow-y-auto" aria-label="Mobile navigation">
   498→                  <Link
   499→                    href="/"
   500→                    className={cn(
   501→                      'px-5 py-3.5 text-base font-medium transition-colors min-h-[44px] flex items-center',
   502→                      pathname === '/'
   503→                        ? 'text-be-brand-yellow bg-white/8 border-l-[3px] border-be-brand-yellow'
   504→                        : 'text-white/90 hover:bg-white/6 hover:text-be-brand-yellow'
   505→                    )}
   506→                    onClick={() => setMobileOpen(false)}
   507→                    aria-current={pathname === '/' ? 'page' : undefined}
   508→                  >

   512→                  {/* Products accordion — with direct /products link */}
