   217→/* ────────────────────────────────────────────
   218→   Section 2: Sticky Product Category Navigator
   219→   ──────────────────────────────────────────── */
   220→
   221→function CategoryNavigator({ activeId }: { activeId: string }) {
   222→  const handleClick = useCallback((id: string) => {
   223→    const el = document.getElementById(id);
   224→    if (el) {
   225→      el.scrollIntoView({ behavior: 'smooth', block: 'start' });
   226→    }
   227→  }, []);
   228→
   229→  return (
   230→    <nav
   231→      aria-label="Product categories"
   232→      className="bg-be-white border-b border-be-grey-150"
   233→    >
   234→      <div className="container-site page-horizontal-padding">
   235→        <div className="flex items-center gap-1 h-[52px] overflow-x-auto scrollbar-none -mx-1 px-1">
   236→          {CATEGORY_NAV_ITEMS.map((item) => {
   237→            const isActive = activeId === item.id;
   238→            return (
   239→              <button
   240→                key={item.id}
   241→                onClick={() => handleClick(item.id)}
   242→                className={`
   243→                  shrink-0 px-4 py-2 text-sm font-medium rounded-md transition-colors duration-200
   244→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-yellow-500
   245→                  ${isActive
   246→                    ? 'bg-be-yellow-500 text-be-charcoal-950'
   247→                    : 'text-be-navy-800 hover:text-be-charcoal-950 hover:bg-be-grey-150'
   248→                  }
   249→                `}
   250→                aria-current={isActive ? 'true' : undefined}
   251→              >
   252→                {item.label}
   253→              </button>
   254→            );
   255→          })}
   256→        </div>
   257→      </div>
   258→    </nav>
   259→  );
   260→}
