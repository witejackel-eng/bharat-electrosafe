   214→function CategoryNavigator({ activeId }: { activeId: string }) {

   220→  }, []);

   224→      aria-label="Product categories"
   225→      className="bg-be-white border-b border-be-grey-150"

   227→      <div className="container-site page-horizontal-padding">
   228→        <div className="flex items-center gap-1 h-[52px] overflow-x-auto scrollbar-none -mx-1 px-1">
   229→          {CATEGORY_NAV_ITEMS.map((item) => {
   230→            const isActive = activeId === item.id;
   231→            return (
   232→              <button
   233→                key={item.id}
   234→                onClick={() => handleClick(item.id)}
   235→                className={`
   236→                  shrink-0 px-4 py-2 text-sm font-medium rounded-md transition-colors duration-200
   237→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-be-yellow-500
   238→                  ${isActive
   239→                    ? 'bg-be-yellow-500 text-be-charcoal-950'
   240→                    : 'text-be-navy-800 hover:text-be-charcoal-950 hover:bg-be-grey-150'
   241→                  }
   242→                `}
   243→                aria-current={isActive ? 'true' : undefined}
   244→              >
   245→                {item.label}
   246→              </button>
   247→            );
   248→          })}
   249→        </div>
   250→      </div>
